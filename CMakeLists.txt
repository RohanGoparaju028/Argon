cmake_minimum_required(VERSION 3.16)
project(Argon VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/runtime.hpp" ARGON_RUNTIME)
configure_file(src/embedded.hpp.in generated/embedded.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/runtime.hpp)
add_executable(argonc src/main.cpp src/frontend.cpp src/compiler.cpp src/project.cpp)
target_include_directories(argonc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(argonc PRIVATE /W4)
else()
  target_compile_options(argonc PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(argon_runtime_tests tests/runtime.cpp)
  target_include_directories(argon_runtime_tests PRIVATE src)
  # These tests deliberately use assert(), including in Release builds.
  if(MSVC)
    target_compile_options(argon_runtime_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(argon_runtime_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME reference_counted_runtime COMMAND argon_runtime_tests)
  add_test(NAME language_and_projects COMMAND ${CMAKE_COMMAND}
    -DCOMPILER=$<TARGET_FILE:argonc> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}
    -DWORK=${CMAKE_CURRENT_BINARY_DIR}/tests -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.cmake)
endif()
find_program(ARGON_NODE node)
if(BUILD_TESTING AND ARGON_NODE)
  add_test(NAME vscode_commands COMMAND ${ARGON_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/editors/vscode/test/tools.test.js)
endif()
find_program(ARGON_VIM vim)
if(BUILD_TESTING AND ARGON_VIM)
  add_test(NAME vim_support COMMAND ${ARGON_VIM} -Nu NONE -n -es -V1 -S ${CMAKE_CURRENT_SOURCE_DIR}/tests/editors.vim)
  set_tests_properties(vim_support PROPERTIES ENVIRONMENT "ARGON_EDITOR_ROOT=${CMAKE_CURRENT_SOURCE_DIR}/editors/vim;ARGON_COMPILER=$<TARGET_FILE:argonc>;ARGON_EXAMPLE=${CMAKE_CURRENT_SOURCE_DIR}/examples/pointers.ar")
endif()
find_program(ARGON_NVIM nvim)
if(BUILD_TESTING AND ARGON_NVIM)
  add_test(NAME neovim_support COMMAND ${ARGON_NVIM} --headless -u NONE -n -S ${CMAKE_CURRENT_SOURCE_DIR}/tests/editors.vim)
  set_tests_properties(neovim_support PROPERTIES ENVIRONMENT "ARGON_EDITOR_ROOT=${CMAKE_CURRENT_SOURCE_DIR}/editors/vim;ARGON_COMPILER=$<TARGET_FILE:argonc>;ARGON_EXAMPLE=${CMAKE_CURRENT_SOURCE_DIR}/examples/pointers.ar")
endif()
find_program(ARGON_LLDB lldb)
if(BUILD_TESTING AND ARGON_LLDB AND NOT WIN32)
  add_test(NAME native_debug_symbols COMMAND ${CMAKE_COMMAND} -DCOMPILER=$<TARGET_FILE:argonc>
    -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DWORK=${CMAKE_CURRENT_BINARY_DIR}
    -DLLDB=${ARGON_LLDB} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/debug-info.cmake)
endif()
install(TARGETS argonc RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/doc/argon)
install(DIRECTORY docs examples DESTINATION share/doc/argon PATTERN ".argon" EXCLUDE)
install(DIRECTORY editors DESTINATION share/argon PATTERN "*.vsix" EXCLUDE PATTERN "test" EXCLUDE)
install(FILES editors/vscode/README.md DESTINATION share/doc/argon/editors/vscode)
install(FILES editors/vim/README.md DESTINATION share/doc/argon/editors/vim)
